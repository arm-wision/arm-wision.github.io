% ----- Column of stages (top to bottom) ---------------------------------
\node[io]    (img)    {Quadrat image \\
                       \small\itshape (raw RGB, $\sim$1800$\times$1800 px)};
\node[stage] (tiles)  [below=of img]    {$4{\times}4$ tile grid \\
                                          {\scriptsize 16 non-overlapping tiles, $448$\,px each}};
\node[emph]  (fwd)    [below=of tiles]  {BioCLIP 2.5 ViT-H/14 \\
                                          {\scriptsize 28 frozen blocks $+$ last 4 blocks, LN, proj.\ unfrozen} \\
                                          {\scriptsize forward at $224$\,px \textbf{and} $336$\,px}};
\node[stage] (agg)    [below=of fwd]    {Per-tile softmax $\to$ mean \\
                                          {\scriptsize unweighted mean over 16 tiles}};
\node[stage] (xres)   [below=of agg]    {Cross-resolution averaging \\
                                          {\scriptsize mean of $224$\,px and $336$\,px probabilities}};
\node[stage] (la)     [below=of xres]   {Class-prior logit adjustment \\
                                          {\scriptsize $\ell'_s = \ell_s - \tau \log \pi_s$,\quad $\tau{=}0.25$}};
\node[stage] (thr)    [below=of la]     {Adaptive threshold \\
                                          {\scriptsize emit $\{s : \bar p(s) > T\}$,\quad $T{=}0.03$,\quad $k \in [2,10]$}};
\node[io]    (out)    [below=of thr]    {Species set \\
                                          \small\itshape (public Macro-F1 $0.418265$)};

% ----- Sequential arrows -----------------------------------------------
\draw[flow] (img)   -- (tiles);
\draw[flow] (tiles) -- (fwd);
\draw[flow] (fwd)   -- (agg);
\draw[flow] (agg)   -- (xres);
\draw[flow] (xres)  -- (la);
\draw[flow] (la)    -- (thr);
\draw[flow] (thr)   -- (out);

% ----- Scope groupings on the left -------------------------------------

  \node[scope, fit=(la) (thr)] (s-post) {};
  \node[scopelabel, anchor=east] at ($(s-post.north west)+(-2pt,-7pt)$) {Post-processing};
