\input{configuration}   % to specify data used in the title page
\input{style-switch}    % if the short thesis layout was chosen, its style gets set here.

\usepackage{xurl}       % allows long URLs to break at any position

\ifLongThesisStyle
  % we only need this for the title page of the long thesis style
  \usepackage[a4paper]{geometry}  
\fi

\usepackage{hyperref}                % defines style of references / links
\hypersetup{
linktocpage,                         % in the table of contents, the numbers serve as links, not the entries
colorlinks  = true,                  % the items are colored instead of colored boxes around them
urlcolor    = cyan,                  % was written in the PDF, all of this can be changed!
linkcolor   = red,
citecolor   = blue
}

\usepackage{datetime}                                % to be able to print month & year on the title page
  \newdateformat{monthonly}{\monthname[\THEMONTH]}
  \newdateformat{monthyeardate}{\monthname[\THEMONTH] \THEYEAR}
\usepackage{amssymb,amsthm,amsmath}                  % standard math packages; often used
\usepackage{graphicx}                                % allows including graphics
\usepackage{natbib}                                  % a specific citation style
\usepackage{floatrow}                                % allows to place a caption next to a figure
  \floatsetup[table]{capposition=top}                 % forces table captions to appear on top.
\usepackage[linesnumbered,ruled,vlined]{algorithm2e} % used for depicting algorithms
\usepackage{booktabs}                                % for tables that actually look nice!
\usepackage{paralist}                                % provides compactitem, a more compact itemize
\usepackage{wrapfig}
\usepackage{needspace}
\usepackage{tabularx}                                % X-columns, used by the experiment-summary tables
\usepackage[section]{placeins}                       % provides \FloatBarrier
\usepackage{tikz}                                    % the report's diagrams are native TikZ
\usetikzlibrary{positioning,arrows.meta,calc,fit,backgrounds,shapes.geometric}
\usepackage{pgfplots}                                % the report's plots are pgfplots
\pgfplotsset{compat=1.18}
\usepgfplotslibrary{groupplots}

% Compatibility shim carried over from the original report preamble:
% colortbl >= v1.0k references \insert@pcolumn for paragraph / X columns,
% which array.sty < 2024 does not define. Alias it to the older token so
% tabularx X-columns compile on mixed package sets.
\makeatletter
\@ifundefined{insert@pcolumn}{\let\insert@pcolumn\insert@column}{}
\makeatother

\begin{document}

\input{macros}                                    % define all your macros here

\input{0-titlepage}                               % define your title page
\input{0-declaration}                             % includes the declaration of authorship
\input{0-acknowledgements}                        % optional acknowledgements
\input{0-abstract}                                % your abstract

\ifLongThesisStyle
  % table of contents (shown only in the long thesis style)
  \renewcommand{\contentsname}{Table of Contents}   % would otherwise just be "Contents",
  \cleardoublepage\tableofcontents\cleardoublepage  % which might sound less nice
  \pagenumbering{arabic}
\fi

% actual report content (ported from the CEUR-ART working note in ../report)
\input{sections/introduction}           % introduction
\input{sections/related_work}           % related work
\input{sections/methodology}            % method
\input{sections/experiments}            % experiments
\input{sections/analysis}               % analysis
\input{sections/phenology}              % seasonal phenology prior (main-body section)
\input{sections/discussion}             % discussion
\input{sections/conclusion}             % conclusion

% Flush any floats still queued before the bibliography begins.
\FloatBarrier

% literature
\setlength{\bibsep}{0pt plus 0.2ex} % tighten inter-reference spacing
\renewcommand*{\bibfont}{\footnotesize} % set references in a smaller font
\bibliographystyle{anuthesis} % or plainnat or whatever
\ifLongThesisStyle
  % this page break is only shown in the long thesis style
  \cleardoublepage\phantomsection
\fi
\bibliography{plantclef-refs}

\end{document}
