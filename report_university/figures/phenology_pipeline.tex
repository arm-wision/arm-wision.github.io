% This figure depicts the SEASONAL PHENOLOGY (Pivot 3) submission, which
% used a richer inference scaffold than the anchor: multi-scale grid tiling,
% ExG vegetation screening, entropy-weighted Bayesian aggregation, and a

% Laid out left-to-right so the figure occupies a short horizontal strip;
% wrap with \resizebox{\linewidth}{!}{...} in the host document.

\begin{tikzpicture}[

% ----- Row of stages (left to right) ---------------------------------
\node[io]    (img)   {Quadrat image};
\node[stage] (tiles) [right=of img]   {Multi-scale tiling \\[1pt] {\tiny $4{\times}4$ grid, scales $\{1.0, 0.8\}$}};
\node[stage] (veg)   [right=of tiles] {ExG veg.\ filter \\[1pt] {\tiny drop tiles $<\!15\%$ green}};
\node[emph]  (fwd)   [right=of veg]   {BioCLIP 2.5 \\ ViT-H/14 \\[1pt] {\tiny last 4 unfrozen; $224{+}336$\,px}};
\node[stage] (agg)   [right=of fwd]   {Bayesian agg. \\ + cross-res \\[1pt] {\tiny $w_t \!\propto\! e^{-H_t}\mathrm{ExG}_t$}};
\node[stage] (phen)  [right=of agg]   {Phenology prior \\[1pt] {\tiny GBIF day-of-year, $\beta{=}1.0$}};
\node[stage] (la)    [right=of phen]  {Logit adjust. \\[1pt] {\tiny $\ell'_s\!=\!\ell_s\!-\!\tau\log\tilde{\pi}_s$, $\tau{=}0.25$}};
\node[stage] (thr)   [right=of la]    {Adaptive thresh. \\[1pt] {\tiny $T{=}0.03$, $k\!\in\![2,10]$}};
\node[io]    (out)   [right=of thr]   {Species set \\[1pt] {\tiny Macro-F1 $0.41346$}};
